cmake_minimum_required(VERSION 3.24)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NacelleProfileTools LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    include(../../cmake/ProfileTools.cmake)
    nacelle_add_profile_support()
    option(NACELLE_BUILD_PROFILE_FUZZER "Build the Clang libFuzzer profile reader" OFF)
    if(NACELLE_BUILD_PROFILE_FUZZER)
        if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            message(FATAL_ERROR "NACELLE_BUILD_PROFILE_FUZZER requires Clang with libFuzzer")
        endif()
        add_executable(profile_reader_fuzz "${NACELLE_PROFILE_ROOT}/tests/profile_reader_fuzz.cpp")
        target_link_libraries(profile_reader_fuzz PRIVATE nacelle_profile_io)
        target_compile_options(profile_reader_fuzz PRIVATE -fsanitize=fuzzer)
        target_link_options(profile_reader_fuzz PRIVATE -fsanitize=fuzzer)
        add_executable(replay_reader_fuzz "${NACELLE_PROFILE_ROOT}/tests/replay_reader_fuzz.cpp")
        target_link_libraries(replay_reader_fuzz PRIVATE nacelle_profile_io)
        target_compile_options(replay_reader_fuzz PRIVATE -fsanitize=fuzzer)
        target_link_options(replay_reader_fuzz PRIVATE -fsanitize=fuzzer)
    endif()
    option(NACELLE_BUILD_PROFILE_TESTS "Build profile capture and format tests" OFF)
    if(NACELLE_BUILD_PROFILE_TESTS)
        enable_testing()
        nacelle_add_profile_tests()
    endif()
endif()

if(NOT EXISTS "${NACELLE_PROFILE_ROOT}/third_party/CLI11/CMakeLists.txt")
    message(FATAL_ERROR "Initialize third_party/CLI11 before building nacelle-dbg")
endif()
set(CLI11_BUILD_TESTS OFF)
set(CLI11_BUILD_EXAMPLES OFF)
add_subdirectory("${NACELLE_PROFILE_ROOT}/third_party/CLI11" "${CMAKE_CURRENT_BINARY_DIR}/cli11" EXCLUDE_FROM_ALL)
add_executable(nacelle-dbg main.cpp)
target_link_libraries(nacelle-dbg PRIVATE nacelle_profile_io CLI11::CLI11)
target_include_directories(nacelle-dbg PRIVATE "${NACELLE_PROFILE_ROOT}/src")
target_compile_features(nacelle-dbg PRIVATE cxx_std_20)
